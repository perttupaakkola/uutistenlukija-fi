{{/* freshness.html — "Juuri nyt" freshness badge + relative timestamp. Usage: {{ partial "freshness.html" . }} Outputs: - If article is < 2h old: Juuri nyt "15 min sitten" - If article is 2–24h old: "3 h sitten" - If article is > 24h old: normal Finnish date ("22.3. klo 14:05") Uses Hugo's built-in time.Duration methods (.Minutes, .Hours). Rendered at build time — accurate to last deploy. */}} {{- $age := now.Sub .Date -}} {{- $ageMin := int $age.Minutes -}} {{- $ageHours := int $age.Hours -}} {{- if lt $ageMin 120 -}} {{/* < 2 hours: badge + relative time */}} Juuri nyt {{- if lt $ageMin 60 -}} {{ $ageMin }} min sitten {{- else -}} {{ $ageHours }} h sitten {{- end -}} {{- else if lt $ageHours 24 -}} {{/* 2–24 hours: hour relative, no badge */}} {{ $ageHours }} h sitten {{- else -}} {{/* > 24 hours: absolute Finnish date */}} {{ .Date | time.Format "2.1. klo 15:04" }} {{- end -}}
