{{- $text := strings.TrimSpace (printf "%v" .) -}} {{- $valid := ge (len $text) 10 -}} {{- $dateText := "" -}} {{- $value := time.AsTime "1970-01-01" -}} {{- if $valid -}} {{- $dateText = substr $text 0 10 -}} {{- if ne (len (findRE `^[0-9]{4}-[0-9]{2}-[0-9]{2}$` $dateText)) 1 -}} {{- $valid = false -}} {{- else -}} {{- $parsed := try (time.AsTime $dateText) -}} {{- if $parsed.Err -}} {{- $valid = false -}} {{- else -}} {{- $value = $parsed.Value -}} {{- end -}} {{- end -}} {{- end -}} {{- return (dict "valid" $valid "text" $dateText "value" $value) -}}
