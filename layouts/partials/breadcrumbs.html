{{/* breadcrumbs.html Renders HTML
breadcrumb + BreadcrumbList JSON-LD. Patterns: Article page: Etusivu › Kategoria › Artikkelin otsikko (truncated ~60) Section page: Etusivu › Kategoria Homepage: (nothing rendered) */}} {{ if not .IsHome }} {{/* ── Build crumb list ─────────────────────────────────────────── */}} {{ $crumbs := slice (dict "name" "Etusivu" "url" .Site.BaseURL) }} {{ if .IsSection }} {{/* Category/section page — one extra crumb for the section itself */}} {{ $crumbs = $crumbs | append (dict "name" .Title "url" .Permalink) }} {{ else if .IsPage }} {{/* Article page — category crumb then article title */}} {{ $cat := "" }}{{ with .Params.categories }}{{ $cat = index . 0 }}{{ end }} {{ if $cat }} {{ $catUrl := printf "%scategories/%s/" .Site.BaseURL ($cat | urlize) }} {{ $crumbs = $crumbs | append (dict "name" $cat "url" $catUrl) }} {{ end }} {{/* Truncate long article titles */}} {{ $title := .Title }} {{ if gt (len $title) 63 }} {{ $title = printf "%s…" (substr $title 0 60) }} {{ end }} {{ $crumbs = $crumbs | append (dict "name" $title "url" .Permalink) }} {{ end }} {{/* ── HTML nav ──────────────────────────────────────────────────── */}}
{{ range $i, $crumb := $crumbs }} {{- if gt $i 0 }}›{{ end -}} {{- if eq $i (sub (len $crumbs) 1) -}} {{ $crumb.name }} {{- else -}} {{ $crumb.name }} {{- end -}} {{ end }}
{{/* ── BreadcrumbList JSON-LD ────────────────────────────────────── */}} {{ $items := slice }} {{ range $i, $crumb := $crumbs }} {{ $items = $items | append (dict "@type" "ListItem" "position" (add $i 1) "name" $crumb.name "item" $crumb.url ) }} {{ end }} {{ $schema := dict "@context" "https://schema.org" "@type" "BreadcrumbList" "itemListElement" $items }} {{ end }}{{/* end if not .IsHome */}}
