{{/* card-image.html — Renders a responsive 16:9 thumbnail for article cards. Parameters (passed as dict via `dict`): .img — full image URL (Unsplash hotlink, Pexels local path, or category JPG) .thumb — thumbnail URL/path (image_thumb frontmatter). Falls back to .img. .alt — alt text .eager — bool: true = loading="eager" fetchpriority="high" (LCP / first card) .hero — bool: true = larger size hints (800w srcset lead) vs 400w card .permalink — article permalink for the wrapping link */}} {{- $img := .img | default "" -}} {{- $thumb := .thumb | default $img -}} {{- $alt := .alt | default "" -}} {{- $eager := .eager | default false -}} {{- $hero := .hero | default false -}} {{- $href := .permalink | default "#" -}} {{- $loading := cond $eager "eager" "lazy" -}} {{- $fetchpri := cond $eager "high" "auto" -}} {{/* Build srcset for Unsplash images (URL params w=N). For Pexels local paths / category JPGs — no srcset (static files). */}} {{- $src := $thumb -}} {{- $srcset := "" -}} {{- $sizes := "" -}} {{- $avifSrcset := "" -}} {{- if hasPrefix $img "https://images.unsplash.com" -}} {{- $base := $img -}} {{- $sep := "?" -}}{{- if in $img "?" -}}{{- $sep = "&" -}}{{- end -}} {{- if $hero -}} {{- $src = printf "%s%sw=800&h=450&fit=crop&fm=webp&q=85" $base $sep -}} {{- $srcset = printf "%s%sw=400&h=225&fit=crop&fm=webp&q=80 400w, %s%sw=800&h=450&fit=crop&fm=webp&q=85 800w, %s%sw=1200&h=675&fit=crop&fm=webp&q=85 1200w" $base $sep $base $sep $base $sep -}} {{- $avifSrcset = printf "%s%sw=400&h=225&fit=crop&fm=avif&q=75 400w, %s%sw=800&h=450&fit=crop&fm=avif&q=80 800w, %s%sw=1200&h=675&fit=crop&fm=avif&q=80 1200w" $base $sep $base $sep $base $sep -}} {{- $sizes = "(max-width: 640px) 100vw, (max-width: 900px) 90vw, 800px" -}} {{- else -}} {{- $src = printf "%s%sw=400&h=225&fit=crop&fm=webp&q=80" $base $sep -}} {{- $srcset = printf "%s%sw=400&h=225&fit=crop&fm=webp&q=80 400w, %s%sw=800&h=450&fit=crop&fm=webp&q=80 800w" $base $sep $base $sep -}} {{- $avifSrcset = printf "%s%sw=400&h=225&fit=crop&fm=avif&q=75 400w, %s%sw=800&h=450&fit=crop&fm=avif&q=75 800w" $base $sep $base $sep -}} {{- $sizes = "(max-width: 640px) calc(100vw - 2rem), 400px" -}} {{- end -}} {{- else if hasPrefix $img "https://images.pexels.com" -}} {{- $base := $img -}} {{- $sep := "?" -}}{{- if in $img "?" -}}{{- $sep = "&" -}}{{- end -}} {{- if $hero -}} {{- $src = printf "%s%sw=800&h=450&fit=crop" $base $sep -}} {{- else -}} {{- $src = printf "%s%sw=400&h=225&fit=crop" $base $sep -}} {{- end -}} {{- if $thumb -}}{{- $src = $thumb -}}{{- end -}} {{- end -}} {{- if $avifSrcset }}{{ end -}} {{- if $srcset }}{{ end -}}
