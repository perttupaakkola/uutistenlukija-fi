{{- /* Single server-side source of truth for all ad-provider paths. Advertising remains dormant until all three activation inputs are present: the explicit feature flag, a provider ID, and a consent revision newer than the dormant v2 preference. Client-side provider access still additionally requires an explicit advertising=true choice for that exact revision. */ -}} {{- $enabled := eq site.Params.ads_enabled true -}} {{- $providerID := trim (site.Params.adsense_id | default "") " " -}} {{- $immutableActivationFloor := 3 -}} {{- $rawConsentRevision := "2" -}} {{- if isset site.Params "ads_consent_revision" -}} {{- $rawConsentRevision = printf "%v" site.Params.ads_consent_revision -}} {{- end -}} {{- $configuredConsentRevision := 0 -}} {{- if gt (len (findRE `^-?[0-9]+$` $rawConsentRevision)) 0 -}} {{- $configuredConsentRevision = int $rawConsentRevision -}} {{- end -}} {{- $rawActivationRevision := printf "%v" $immutableActivationFloor -}} {{- if isset site.Params "ads_activation_revision" -}} {{- $rawActivationRevision = printf "%v" site.Params.ads_activation_revision -}} {{- end -}} {{- $activationRevision := 0 -}} {{- if gt (len (findRE `^-?[0-9]+$` $rawActivationRevision)) 0 -}} {{- $activationRevision = int $rawActivationRevision -}} {{- end -}} {{- $activationRequested := and $enabled (ne $providerID "") -}} {{- $activationFloorValid := ge $activationRevision $immutableActivationFloor -}} {{- $consentRevisionValid := and (ge $configuredConsentRevision $immutableActivationFloor) (ge $configuredConsentRevision $activationRevision) -}} {{- $serverEligible := and $activationRequested $activationFloorValid $consentRevisionValid -}} {{- $currentConsentRevision := 2 -}} {{- if $activationRequested -}} {{- $currentConsentRevision = $immutableActivationFloor -}} {{- if gt $configuredConsentRevision $immutableActivationFloor -}} {{- $currentConsentRevision = $configuredConsentRevision -}} {{- end -}} {{- end -}} {{- return (dict "enabled" $enabled "providerID" $providerID "consentRevision" $currentConsentRevision "configuredConsentRevision" $configuredConsentRevision "activationRevision" $activationRevision "immutableActivationFloor" $immutableActivationFloor "activationRequested" $activationRequested "activationFloorValid" $activationFloorValid "serverEligible" $serverEligible ) -}}
