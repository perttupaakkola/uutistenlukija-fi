{{/* freshness-label.html — relative time labels in Finnish. Usage: {{ partial "freshness-label.html" .Date }} < 1h → Juuri nyt < 6h → "X tuntia sitten" Same day → "Tänään klo HH:MM" Yesterday → "Eilen klo HH:MM" Older → "D.M. klo HH:MM" */}} {{- $date := . -}} {{- $now := now -}} {{- $diffSec := math.Round (sub $now.Unix $date.Unix) -}} {{- $diffHours := math.Round (div $diffSec 3600) -}} {{- $time := $date | time.Format "15:04" -}} {{- $nowDay := $now | time.Format "2006-01-02" -}} {{- $dateDay := $date | time.Format "2006-01-02" -}} {{- $yesterdayUnix := sub $now.Unix 86400 -}} {{- $yesterday := time $yesterdayUnix -}} {{- $yesterdayDay := $yesterday | time.Format "2006-01-02" -}} {{- if lt $diffSec 3600.0 -}} Juuri nyt {{- else if lt $diffSec 21600.0 -}} {{ int $diffHours }} tuntia sitten {{- else if eq $nowDay $dateDay -}} Tänään klo {{ $time }} {{- else if eq $yesterdayDay $dateDay -}} Eilen klo {{ $time }} {{- else -}} {{ $date | time.Format "2.1. klo 15:04" }} {{- end -}}
