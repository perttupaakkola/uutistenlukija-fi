{{/* responsive-image.html — Responsive image with CDN optimization + srcset For Unsplash/Pexels: renders with WebP + JPEG sources at correct crop dimensions. For local/static images: plain . Parameters: - src: image URL - alt: alt text - width: display width (400 for cards, 800 for lead, 1200 for hero) - height: display height (225 for cards, 450 for lead, 675 for hero) - class: CSS class - loading: lazy or eager (default: lazy) - fetchpriority: high / auto (default: auto; set high for LCP element) - sizes: responsive sizes string (optional) */}} {{ $src := .src }} {{ $alt := .alt | default "" }} {{ $width := .width | default 400 }} {{ $height := .height | default 225 }} {{ $class := .class | default "" }} {{ $loading := .loading | default "lazy" }} {{ $fetchpriority := .fetchpriority | default "" }} {{ $sizes := .sizes | default "" }} {{ $isUnsplash := hasPrefix $src "https://images.unsplash.com" }} {{ $isPexels := hasPrefix $src "https://images.pexels.com" }} {{ $sep := "?" }}{{ if in $src "?" }}{{ $sep = "&" }}{{ end }} {{ if $isUnsplash }} {{ $webp := printf "%s%sw=%d&h=%d&fit=crop&fm=webp&q=80" $src $sep $width $height }} {{ $jpeg := printf "%s%sw=%d&h=%d&fit=crop&fm=jpg&q=85" $src $sep $width $height }} {{ else if $isPexels }} {{ $sized := printf "%s%sw=%d&h=%d&fit=crop" $src $sep $width $height }} {{ else }} {{ end }}
